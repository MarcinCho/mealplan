# Etap 2: Rdzeń Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Domownicy mogą dodawać przepisy ręcznie, rozpisać tydzień (przepis lub wolny tekst, kucharz, porcje) i wygenerować z planu listę zakupów; admin zarządza rolami i członkami.

**Architecture:** Nowe pakiety domenowe `units`, `recipe`, `plan`, `shopping` nie znają HTTP ani SQL; mówią do repozytoriów przez małe interfejsy, które spełnia `*store.Store`. Każde zapytanie repozytorium dostaje `householdID` jako obowiązkowy parametr. `httpapi` dostaje strukturę `Deps` z usługami. Frontend dokłada widoki Przepisy, Plan i Zakupy w istniejącej powłoce.

**Tech Stack:** Go 1.26 (`net/http` ServeMux), `database/sql` (SQLite `modernc.org/sqlite`, Postgres `pgx`), Vue 3 + Vuetify 3 + Pinia + Vitest.

**Spec:** `docs/superpowers/specs/2026-11-01-mealplan-design.md` (sekcje 4, 5, 7.1, 7.2, 8, 12).

## Global Constraints

- Wszystkie ścieżki API pod `/api/v1`; błędy w formacie `{code, message, details}`.
- Każdy nowy route musi być równocześnie opisany w `api/openapi.yaml` (test `TestRoutesMatchOpenAPI` pilnuje zgodności) w tym samym zadaniu.
- Każde zapytanie do tabel domenowych zawiera `household_id`; testy potwierdzają, że gospodarstwo A nie odczyta ani nie zmieni danych B.
- SQL przenośny SQLite/Postgres: placeholdery `?` (przez `Rebind`), czasy jako `BIGINT` (sekundy unix), daty planu jako `TEXT` `YYYY-MM-DD`, flagi jako `INTEGER` 0/1, ilości jako `DOUBLE PRECISION`. W plikach migracji nie wolno używać średnika poza końcem instrukcji (migrator dzieli po `;`).
- Jednostki: obsługiwane konwersje tylko g↔kg i ml↔l; konwersji międzygrupowych (np. łyżka↔gram) nie robimy.
- Porcje: `ilość × porcje_w_planie ÷ porcje_bazowe`.
- Tekst interfejsu po polsku, w `web/src/messages.ts`. Komunikaty błędów serwera po angielsku, mapowane po `code`.
- Wszyscy członkowie gospodarstwa mogą edytować przepisy, plan i listę zakupów; tylko `admin` zarządza członkami i zaproszeniami.
- Commity w stylu Etapu 1 (`feat(scope): ...`); każdy commit kończy się linią `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.

```go
// sqliteDSN makes sure foreign keys are enforced; SQLite leaves them off by default and the
// setting is per connection, so it has to travel with the DSN.
func sqliteDSN(dsn string) string {
	if strings.Contains(dsn, "foreign_keys") {
		return dsn
	}
	sep := "?"
	if strings.Contains(dsn, "?") {
		sep = "&"
	}
	return dsn + sep + "_pragma=foreign_keys(1)"
}
```

```go
// Package units normalizes ingredient names and units and scales quantities.
// Only g<->kg and ml<->l are converted; other units are compared as written.
package units

import (
	"errors"
	"math"
	"strings"
)

var ErrBadServings = errors.New("units: servings must be positive")

var aliases = map[string]string{
	"szt": "szt", "szt.": "szt", "sztuka": "szt", "sztuki": "szt", "sztuk": "szt",
	"gram": "g", "gramy": "g", "gramów": "g",
	"kilogram": "kg", "kilogramy": "kg",
	"litr": "l", "litry": "l", "mililitr": "ml", "mililitry": "ml",
}

// NormalizeName lowercases, trims and collapses whitespace so that equal ingredients match.
func NormalizeName(s string) string {
	return strings.Join(strings.Fields(strings.ToLower(s)), " ")
}

// CanonUnit lowercases a unit and maps common spellings to one form.
func CanonUnit(u string) string {
	u = strings.ToLower(strings.TrimSpace(u))
	if a, ok := aliases[u]; ok {
		return a
	}
	return u
}

// ToBase converts kg to g and l to ml; other units are returned canonicalized and unchanged.
func ToBase(amount float64, unit string) (float64, string) {
	switch u := CanonUnit(unit); u {
	case "kg":
		return amount * 1000, "g"
	case "l":
		return amount * 1000, "ml"
	default:
		return amount, u
	}
}

// Humanize is the inverse of ToBase for display: 1500 g becomes 1.5 kg.
func Humanize(amount float64, unit string) (float64, string) {
	switch {
	case unit == "g" && amount >= 1000:
		return Round(amount / 1000), "kg"
	case unit == "ml" && amount >= 1000:
		return Round(amount / 1000), "l"
	default:
		return Round(amount), unit
	}
}

// Round keeps three decimals so float noise never reaches the user.
func Round(a float64) float64 { return math.Round(a*1000) / 1000 }

// Scale returns amount * servings / base, rounded.
func Scale(amount float64, servings, base int) (float64, error) {
	if servings <= 0 || base <= 0 {
		return 0, ErrBadServings
	}
	return Round(amount * float64(servings) / float64(base)), nil
}
```

```go
import (
	"context"
```

```go
func (s *Store) CreateRecipe(ctx context.Context, r domain.Recipe) error {
	return s.inTx(ctx, func(tx *sql.Tx) error {
		if err := s.exec(ctx, tx, `INSERT INTO recipe (id, household_id, title, base_servings, source_url, created_at, updated_at) VALUES (?, ?, ?, ?, ?, ?, ?)`,
			r.ID, r.HouseholdID, r.Title, r.BaseServings, r.SourceURL, ts(r.CreatedAt), ts(r.UpdatedAt)); err != nil {
			return err
		}
		return s.insertRecipeChildren(ctx, tx, r)
	})
}

func (s *Store) UpdateRecipe(ctx context.Context, r domain.Recipe) error {
	return s.inTx(ctx, func(tx *sql.Tx) error {
		res, err := tx.ExecContext(ctx, s.db.Rebind(`UPDATE recipe SET title = ?, base_servings = ?, source_url = ?, updated_at = ? WHERE id = ? AND household_id = ?`),
			r.Title, r.BaseServings, r.SourceURL, ts(r.UpdatedAt), r.ID, r.HouseholdID)
		if err != nil {
			return mapErr(err)
		}
		if n, _ := res.RowsAffected(); n == 0 {
			return ErrNotFound
		}
		for _, tbl := range []string{"ingredient", "step", "recipe_tag"} {
			if err := s.exec(ctx, tx, `DELETE FROM `+tbl+` WHERE recipe_id = ?`, r.ID); err != nil {
				return err
			}
		}
		return s.insertRecipeChildren(ctx, tx, r)
	})
}

func (s *Store) insertRecipeChildren(ctx context.Context, tx *sql.Tx, r domain.Recipe) error {
	for i, ing := range r.Ingredients {
		var q any
		if ing.Quantity != nil {
			q = *ing.Quantity
		}
		if err := s.exec(ctx, tx, `INSERT INTO ingredient (recipe_id, position, name, normalized_name, quantity, unit) VALUES (?, ?, ?, ?, ?, ?)`,
			r.ID, i, ing.Name, ing.NormalizedName, q, ing.Unit); err != nil {
			return err
		}
	}
	for i, text := range r.Steps {
		if err := s.exec(ctx, tx, `INSERT INTO step (recipe_id, position, text) VALUES (?, ?, ?)`, r.ID, i, text); err != nil {
			return err
		}
	}
	for _, tag := range r.Tags {
		if err := s.exec(ctx, tx, `INSERT INTO recipe_tag (recipe_id, tag) VALUES (?, ?)`, r.ID, tag); err != nil {
			return err
		}
	}
	return nil
}

func (s *Store) RecipeByID(ctx context.Context, householdID, id string) (domain.Recipe, error) {
	var r domain.Recipe
	var created, updated int64
	err := s.db.QueryRowContext(ctx, s.db.Rebind(
		`SELECT id, household_id, title, base_servings, source_url, created_at, updated_at FROM recipe WHERE id = ? AND household_id = ?`),
		id, householdID).Scan(&r.ID, &r.HouseholdID, &r.Title, &r.BaseServings, &r.SourceURL, &created, &updated)
	if err != nil {
		return domain.Recipe{}, mapErr(err)
	}
	r.CreatedAt, r.UpdatedAt = fromTS(created), fromTS(updated)

	rows, err := s.db.QueryContext(ctx, s.db.Rebind(`SELECT name, normalized_name, quantity, unit FROM ingredient WHERE recipe_id = ? ORDER BY position`), id)
	if err != nil {
		return domain.Recipe{}, err
	}
	for rows.Next() {
		var ing domain.Ingredient
		var q sql.NullFloat64
		if err := rows.Scan(&ing.Name, &ing.NormalizedName, &q, &ing.Unit); err != nil {
			_ = rows.Close()
			return domain.Recipe{}, err
		}
		if q.Valid {
			v := q.Float64
			ing.Quantity = &v
		}
		r.Ingredients = append(r.Ingredients, ing)
	}
	if err := rows.Close(); err != nil {
		return domain.Recipe{}, err
	}

	if r.Steps, err = s.stringCol(ctx, `SELECT text FROM step WHERE recipe_id = ? ORDER BY position`, id); err != nil {
		return domain.Recipe{}, err
	}
	if r.Tags, err = s.stringCol(ctx, `SELECT tag FROM recipe_tag WHERE recipe_id = ? ORDER BY tag`, id); err != nil {
		return domain.Recipe{}, err
	}
	return r, nil
}

func (s *Store) stringCol(ctx context.Context, q string, args ...any) ([]string, error) {
	rows, err := s.db.QueryContext(ctx, s.db.Rebind(q), args...)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []string
	for rows.Next() {
		var v string
		if err := rows.Scan(&v); err != nil {
			return nil, err
		}
		out = append(out, v)
	}
	return out, rows.Err()
}

// ListRecipes returns summaries (title, servings, source, tags) ordered by title.
func (s *Store) ListRecipes(ctx context.Context, householdID string) ([]domain.Recipe, error) {
	rows, err := s.db.QueryContext(ctx, s.db.Rebind(
		`SELECT id, household_id, title, base_servings, source_url, created_at, updated_at FROM recipe WHERE household_id = ? ORDER BY title, id`), householdID)
	if err != nil {
		return nil, err
	}
	var out []domain.Recipe
	idx := map[string]int{}
	for rows.Next() {
		var r domain.Recipe
		var created, updated int64
		if err := rows.Scan(&r.ID, &r.HouseholdID, &r.Title, &r.BaseServings, &r.SourceURL, &created, &updated); err != nil {
			_ = rows.Close()
			return nil, err
		}
		r.CreatedAt, r.UpdatedAt = fromTS(created), fromTS(updated)
		idx[r.ID] = len(out)
		out = append(out, r)
	}
	if err := rows.Close(); err != nil {
		return nil, err
	}
	trows, err := s.db.QueryContext(ctx, s.db.Rebind(
		`SELECT t.recipe_id, t.tag FROM recipe_tag t JOIN recipe r ON r.id = t.recipe_id WHERE r.household_id = ? ORDER BY t.tag`), householdID)
	if err != nil {
		return nil, err
	}
	defer trows.Close()
	for trows.Next() {
		var id, tag string
		if err := trows.Scan(&id, &tag); err != nil {
			return nil, err
		}
		if i, ok := idx[id]; ok {
			out[i].Tags = append(out[i].Tags, tag)
		}
	}
	return out, trows.Err()
}

func (s *Store) DeleteRecipe(ctx context.Context, householdID, id string) error {
	return s.inTx(ctx, func(tx *sql.Tx) error {
		var n int
		if err := tx.QueryRowContext(ctx, s.db.Rebind(`SELECT COUNT(*) FROM recipe WHERE id = ? AND household_id = ?`), id, householdID).Scan(&n); err != nil {
			return err
		}
		if n == 0 {
			return ErrNotFound
		}
		if err := tx.QueryRowContext(ctx, s.db.Rebind(`SELECT COUNT(*) FROM plan_entry WHERE recipe_id = ? AND household_id = ?`), id, householdID).Scan(&n); err != nil {
			return err
		}
		if n > 0 {
			return ErrInUse
		}
		return s.exec(ctx, tx, `DELETE FROM recipe WHERE id = ? AND household_id = ?`, id, householdID)
	})
}
```

```go
// Package recipe holds recipe use cases: validation, normalization and persistence calls.
package recipe

import (
	"context"
	"fmt"
	"math"
	"net/url"
	"strings"
	"time"
	"unicode/utf8"

	"mealplan/internal/auth"
	"mealplan/internal/domain"
	"mealplan/internal/units"
)

const (
	maxTitle       = 200
	maxServings    = 100
	maxIngredients = 200
	maxSteps       = 100
	maxStepLen     = 2000
	maxTags        = 20
	maxTagLen      = 40
	maxUnitLen     = 30
	maxQuantity    = 1e6
	maxURL         = 2000
)

type Repo interface {
	CreateRecipe(ctx context.Context, r domain.Recipe) error
	UpdateRecipe(ctx context.Context, r domain.Recipe) error
	RecipeByID(ctx context.Context, householdID, id string) (domain.Recipe, error)
	ListRecipes(ctx context.Context, householdID string) ([]domain.Recipe, error)
	DeleteRecipe(ctx context.Context, householdID, id string) error
}

type IngredientInput struct {
	Name     string
	Quantity *float64
	Unit     string
}

type Input struct {
	Title        string
	BaseServings int
	SourceURL    string
	Tags         []string
	Ingredients  []IngredientInput
	Steps        []string
}

type Service struct {
	repo Repo
	now  func() time.Time
}

func New(repo Repo, now func() time.Time) *Service {
	if now == nil {
		now = time.Now
	}
	return &Service{repo: repo, now: now}
}

func invalid(format string, a ...any) error {
	return fmt.Errorf("%w: "+format, append([]any{domain.ErrInvalid}, a...)...)
}

// build validates the input and returns a recipe with normalized fields (no ID, owner or times).
func build(in Input) (domain.Recipe, error) {
	r := domain.Recipe{Title: strings.TrimSpace(in.Title), BaseServings: in.BaseServings, SourceURL: strings.TrimSpace(in.SourceURL)}
	if n := utf8.RuneCountInString(r.Title); n == 0 || n > maxTitle {
		return r, invalid("title must be 1 to %d characters", maxTitle)
	}
	if r.BaseServings < 1 || r.BaseServings > maxServings {
		return r, invalid("baseServings must be 1 to %d", maxServings)
	}
	if r.SourceURL != "" {
		u, err := url.Parse(r.SourceURL)
		if err != nil || len(r.SourceURL) > maxURL || (u.Scheme != "http" && u.Scheme != "https") || u.Host == "" {
			return r, invalid("sourceUrl must be an http or https address")
		}
	}
	if len(in.Ingredients) > maxIngredients {
		return r, invalid("at most %d ingredients", maxIngredients)
	}
	for _, ing := range in.Ingredients {
		name := strings.TrimSpace(ing.Name)
		unit := units.CanonUnit(ing.Unit)
		if n := utf8.RuneCountInString(name); n == 0 || n > maxTitle {
			return r, invalid("ingredient name must be 1 to %d characters", maxTitle)
		}
		if utf8.RuneCountInString(unit) > maxUnitLen {
			return r, invalid("unit must be at most %d characters", maxUnitLen)
		}
		if ing.Quantity != nil {
			v := *ing.Quantity
			if math.IsNaN(v) || math.IsInf(v, 0) || v <= 0 || v > maxQuantity {
				return r, invalid("quantity must be greater than 0 and at most %v", maxQuantity)
			}
		}
		r.Ingredients = append(r.Ingredients, domain.Ingredient{Name: name, NormalizedName: units.NormalizeName(name), Quantity: ing.Quantity, Unit: unit})
	}
	if len(in.Steps) > maxSteps {
		return r, invalid("at most %d steps", maxSteps)
	}
	for _, s := range in.Steps {
		s = strings.TrimSpace(s)
		if n := utf8.RuneCountInString(s); n == 0 || n > maxStepLen {
			return r, invalid("each step must be 1 to %d characters", maxStepLen)
		}
		r.Steps = append(r.Steps, s)
	}
	if len(in.Tags) > maxTags {
		return r, invalid("at most %d tags", maxTags)
	}
	seen := map[string]bool{}
	for _, tag := range in.Tags {
		tag = units.NormalizeName(tag)
		if n := utf8.RuneCountInString(tag); n == 0 || n > maxTagLen {
			return r, invalid("each tag must be 1 to %d characters", maxTagLen)
		}
		if !seen[tag] {
			seen[tag] = true
			r.Tags = append(r.Tags, tag)
		}
	}
	return r, nil
}

func (s *Service) Create(ctx context.Context, householdID string, in Input) (domain.Recipe, error) {
	r, err := build(in)
	if err != nil {
		return domain.Recipe{}, err
	}
	r.ID, r.HouseholdID = auth.NewID(), householdID
	r.CreatedAt = s.now().UTC()
	r.UpdatedAt = r.CreatedAt
	if err := s.repo.CreateRecipe(ctx, r); err != nil {
		return domain.Recipe{}, err
	}
	return r, nil
}

func (s *Service) Update(ctx context.Context, householdID, id string, in Input) (domain.Recipe, error) {
	r, err := build(in)
	if err != nil {
```

```go
// Package plan holds the weekly meal plan use cases.
package plan

import (
	"fmt"
	"time"

	"mealplan/internal/domain"
)

const dateLayout = "2006-01-02"

// ValidWeek accepts only a YYYY-MM-DD string that is a Monday and returns it as UTC midnight.
func ValidWeek(week string) (time.Time, error) {
	t, err := time.Parse(dateLayout, week)
	if err != nil || t.Format(dateLayout) != week {
		return time.Time{}, fmt.Errorf("%w: week must be a YYYY-MM-DD date", domain.ErrInvalid)
	}
	if t.Weekday() != time.Monday {
		return time.Time{}, fmt.Errorf("%w: week must start on a Monday", domain.ErrInvalid)
	}
	return t, nil
}

// WeekDates returns the seven dates of the week starting on the given Monday.
func WeekDates(week string) ([]string, error) {
	start, err := ValidWeek(week)
	if err != nil {
		return nil, err
	}
	out := make([]string, 7)
	for i := range out {
		out[i] = start.AddDate(0, 0, i).Format(dateLayout)
	}
	return out, nil
}
```

```go
package store

import (
	"context"
	"database/sql"

	"mealplan/internal/domain"
)

const planCols = `id, household_id, date, slot, recipe_id, free_text, servings, cook_user_id`

func nullable(s string) any {
	if s == "" {
		return nil
	}
	return s
}

func scanEntry(r scanner) (domain.PlanEntry, error) {
	var e domain.PlanEntry
	var slot string
	var recipe, cook sql.NullString
	if err := r.Scan(&e.ID, &e.HouseholdID, &e.Date, &slot, &recipe, &e.FreeText, &e.Servings, &cook); err != nil {
		return domain.PlanEntry{}, mapErr(err)
	}
	e.Slot = domain.Slot(slot)
	e.RecipeID, e.CookUserID = recipe.String, cook.String
	return e, nil
}

// checkRefs verifies, inside the transaction, that the recipe and the cook belong to the household.
func (s *Store) checkRefs(ctx context.Context, tx *sql.Tx, e domain.PlanEntry) error {
	if e.RecipeID != "" {
		var n int
		if err := tx.QueryRowContext(ctx, s.db.Rebind(`SELECT COUNT(*) FROM recipe WHERE id = ? AND household_id = ?`), e.RecipeID, e.HouseholdID).Scan(&n); err != nil {
			return err
		}
		if n == 0 {
			return ErrNotFound
		}
	}
	if e.CookUserID != "" {
		var n int
		if err := tx.QueryRowContext(ctx, s.db.Rebind(`SELECT COUNT(*) FROM membership WHERE user_id = ? AND household_id = ?`), e.CookUserID, e.HouseholdID).Scan(&n); err != nil {
			return err
		}
		if n == 0 {
			return ErrNotFound
		}
	}
	return nil
}

func (s *Store) insertEntry(ctx context.Context, tx *sql.Tx, e domain.PlanEntry) error {
	if err := s.checkRefs(ctx, tx, e); err != nil {
		return err
	}
	return s.exec(ctx, tx, `INSERT INTO plan_entry (`+planCols+`) VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
		e.ID, e.HouseholdID, e.Date, string(e.Slot), nullable(e.RecipeID), e.FreeText, e.Servings, nullable(e.CookUserID))
}

func (s *Store) InsertPlanEntry(ctx context.Context, e domain.PlanEntry) error {
	return s.inTx(ctx, func(tx *sql.Tx) error { return s.insertEntry(ctx, tx, e) })
}

func (s *Store) UpdatePlanEntry(ctx context.Context, e domain.PlanEntry) error {
	return s.inTx(ctx, func(tx *sql.Tx) error {
		if err := s.checkRefs(ctx, tx, e); err != nil {
			return err
		}
		res, err := tx.ExecContext(ctx, s.db.Rebind(
			`UPDATE plan_entry SET date = ?, slot = ?, recipe_id = ?, free_text = ?, servings = ?, cook_user_id = ? WHERE id = ? AND household_id = ?`),
			e.Date, string(e.Slot), nullable(e.RecipeID), e.FreeText, e.Servings, nullable(e.CookUserID), e.ID, e.HouseholdID)
		if err != nil {
			return mapErr(err)
		}
		if n, _ := res.RowsAffected(); n == 0 {
			return ErrNotFound
		}
		return nil
	})
}

func (s *Store) DeletePlanEntry(ctx context.Context, householdID, id string) error {
	res, err := s.db.ExecContext(ctx, s.db.Rebind(`DELETE FROM plan_entry WHERE id = ? AND household_id = ?`), id, householdID)
	if err != nil {
		return mapErr(err)
	}
	if n, _ := res.RowsAffected(); n == 0 {
		return ErrNotFound
	}
	return nil
}

func (s *Store) PlanEntryByID(ctx context.Context, householdID, id string) (domain.PlanEntry, error) {
	return scanEntry(s.db.QueryRowContext(ctx, s.db.Rebind(`SELECT `+planCols+` FROM plan_entry WHERE id = ? AND household_id = ?`), id, householdID))
}

// PlanEntries returns entries with fromDate <= date <= toDate (YYYY-MM-DD compares lexically).
func (s *Store) PlanEntries(ctx context.Context, householdID, fromDate, toDate string) ([]domain.PlanEntry, error) {
	rows, err := s.db.QueryContext(ctx, s.db.Rebind(
		`SELECT `+planCols+` FROM plan_entry WHERE household_id = ? AND date >= ? AND date <= ? ORDER BY date, slot, id`),
		householdID, fromDate, toDate)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []domain.PlanEntry
	for rows.Next() {
		e, err := scanEntry(rows)
		if err != nil {
			return nil, err
		}
		out = append(out, e)
	}
	return out, rows.Err()
}

// CopyPlanEntries inserts the batch atomically. It refuses with ErrConflict when the household
// already has entries on any of the batch's dates' week range (min..max date of the batch).
func (s *Store) CopyPlanEntries(ctx context.Context, householdID string, entries []domain.PlanEntry) error {
	if len(entries) == 0 {
		return nil
	}
	lo, hi := entries[0].Date, entries[0].Date
	for _, e := range entries {
		if e.Date < lo {
			lo = e.Date
		}
		if e.Date > hi {
			hi = e.Date
		}
	}
	return s.inTx(ctx, func(tx *sql.Tx) error {
		var n int
		if err := tx.QueryRowContext(ctx, s.db.Rebind(`SELECT COUNT(*) FROM plan_entry WHERE household_id = ? AND date >= ? AND date <= ?`), householdID, lo, hi).Scan(&n); err != nil {
			return err
		}
		if n > 0 {
			return ErrConflict
		}
		for _, e := range entries {
			e.HouseholdID = householdID
			if err := s.insertEntry(ctx, tx, e); err != nil {
				return err
			}
		}
		return nil
	})
}
```

```go
package plan

import (
	"context"
	"fmt"
	"strings"
	"unicode/utf8"

	"mealplan/internal/auth"
	"mealplan/internal/domain"
	"mealplan/internal/store"
)

const (
	maxFreeText = 300
	maxServings = 100
)

type Repo interface {
	InsertPlanEntry(ctx context.Context, e domain.PlanEntry) error
	UpdatePlanEntry(ctx context.Context, e domain.PlanEntry) error
	DeletePlanEntry(ctx context.Context, householdID, id string) error
	PlanEntryByID(ctx context.Context, householdID, id string) (domain.PlanEntry, error)
	PlanEntries(ctx context.Context, householdID, fromDate, toDate string) ([]domain.PlanEntry, error)
	CopyPlanEntries(ctx context.Context, householdID string, entries []domain.PlanEntry) error
}

type RecipeLookup interface {
	RecipeByID(ctx context.Context, householdID, id string) (domain.Recipe, error)
}

type EntryInput struct {
	Date       string
	Slot       domain.Slot
	RecipeID   string
	FreeText   string
	Servings   int // 0 = the recipe's base servings (1 for free text)
	CookUserID string
}

type Service struct {
	repo    Repo
	recipes RecipeLookup
}

func New(repo Repo, recipes RecipeLookup) *Service { return &Service{repo: repo, recipes: recipes} }

func invalid(format string, a ...any) error {
	return fmt.Errorf("%w: "+format, append([]any{domain.ErrInvalid}, a...)...)
}

// entry validates the input against the week and resolves defaults.
func (s *Service) entry(ctx context.Context, householdID, week string, in EntryInput) (domain.PlanEntry, error) {
	dates, err := WeekDates(week)
	if err != nil {
		return domain.PlanEntry{}, err
	}
	inWeek := false
	for _, d := range dates {
		if d == in.Date {
			inWeek = true
		}
	}
	if !inWeek {
		return domain.PlanEntry{}, invalid("date must fall inside the week starting %s", week)
	}
	if !in.Slot.Valid() {
		return domain.PlanEntry{}, invalid("slot must be breakfast, lunch, dinner or snack")
	}
	text := strings.TrimSpace(in.FreeText)
	if (in.RecipeID == "") == (text == "") {
		return domain.PlanEntry{}, invalid("provide exactly one of recipeId and freeText")
	}
	if utf8.RuneCountInString(text) > maxFreeText {
		return domain.PlanEntry{}, invalid("freeText must be at most %d characters", maxFreeText)
	}
	if in.Servings < 0 || in.Servings > maxServings {
		return domain.PlanEntry{}, invalid("servings must be 1 to %d", maxServings)
	}
	servings := in.Servings
	if in.RecipeID != "" {
		r, err := s.recipes.RecipeByID(ctx, householdID, in.RecipeID)
		if err != nil {
			return domain.PlanEntry{}, err
		}
		if servings == 0 {
			servings = r.BaseServings
		}
	} else if servings == 0 {
		servings = 1
	}
	return domain.PlanEntry{HouseholdID: householdID, Date: in.Date, Slot: in.Slot, RecipeID: in.RecipeID,
		FreeText: text, Servings: servings, CookUserID: in.CookUserID}, nil
}

func (s *Service) Week(ctx context.Context, householdID, week string) ([]domain.PlanEntry, error) {
	dates, err := WeekDates(week)
	if err != nil {
		return nil, err
	}
	return s.repo.PlanEntries(ctx, householdID, dates[0], dates[6])
}

func (s *Service) Add(ctx context.Context, householdID, week string, in EntryInput) (domain.PlanEntry, error) {
	e, err := s.entry(ctx, householdID, week, in)
	if err != nil {
		return domain.PlanEntry{}, err
	}
	e.ID = auth.NewID()
	if err := s.repo.InsertPlanEntry(ctx, e); err != nil {
		return domain.PlanEntry{}, err
	}
	return e, nil
}

func (s *Service) Edit(ctx context.Context, householdID, week, id string, in EntryInput) (domain.PlanEntry, error) {
	e, err := s.entry(ctx, householdID, week, in)
	if err != nil {
		return domain.PlanEntry{}, err
	}
	e.ID = id
	if err := s.repo.UpdatePlanEntry(ctx, e); err != nil {
		return domain.PlanEntry{}, err
	}
	return e, nil
}

func (s *Service) Remove(ctx context.Context, householdID, week, id string) error {
	if _, err := ValidWeek(week); err != nil {
		return err
	}
	return s.repo.DeletePlanEntry(ctx, householdID, id)
}

// Copy duplicates every entry of fromWeek into toWeek, shifted by whole weeks, and returns how many
// entries it created. The target week must be empty.
func (s *Service) Copy(ctx context.Context, householdID, fromWeek, toWeek string) (int, error) {
	from, err := ValidWeek(fromWeek)
	if err != nil {
		return 0, err
	}
	to, err := ValidWeek(toWeek)
	if err != nil {
		return 0, err
	}
	if fromWeek == toWeek {
		return 0, invalid("source and target week must differ")
	}
	src, err := s.Week(ctx, householdID, fromWeek)
	if err != nil {
		return 0, err
	}
	existing, err := s.Week(ctx, householdID, toWeek)
	if err != nil {
		return 0, err
	}
	if len(existing) > 0 {
		return 0, store.ErrConflict
	}
	shift := int(to.Sub(from).Hours() / 24)
	out := make([]domain.PlanEntry, 0, len(src))
	for _, e := range src {
		d, _ := time.Parse(dateLayout, e.Date)
		e.ID = auth.NewID()
		e.Date = d.AddDate(0, 0, shift).Format(dateLayout)
		out = append(out, e)
	}
	if err := s.repo.CopyPlanEntries(ctx, householdID, out); err != nil {
		return 0, err
	}
	return len(out), nil
}
```

```go
package httpapi

import (
	"net/http"

	"mealplan/internal/domain"
	"mealplan/internal/plan"
)

type entryJSON struct {
	ID         string  `json:"id"`
	Date       string  `json:"date"`
	Slot       string  `json:"slot"`
	RecipeID   *string `json:"recipeId"`
	FreeText   string  `json:"freeText"`
	Servings   int     `json:"servings"`
	CookUserID *string `json:"cookUserId"`
}

func strPtr(s string) *string {
	if s == "" {
		return nil
	}
	return &s
}

func toEntryJSON(e domain.PlanEntry) entryJSON {
	return entryJSON{ID: e.ID, Date: e.Date, Slot: string(e.Slot), RecipeID: strPtr(e.RecipeID),
		FreeText: e.FreeText, Servings: e.Servings, CookUserID: strPtr(e.CookUserID)}
}

type entryInJSON struct {
	Date       string  `json:"date"`
	Slot       string  `json:"slot"`
	RecipeID   *string `json:"recipeId"`
	FreeText   string  `json:"freeText"`
	Servings   int     `json:"servings"`
	CookUserID *string `json:"cookUserId"`
}

func (in entryInJSON) toInput() plan.EntryInput {
	out := plan.EntryInput{Date: in.Date, Slot: domain.Slot(in.Slot), FreeText: in.FreeText, Servings: in.Servings}
	if in.RecipeID != nil {
		out.RecipeID = *in.RecipeID
	}
	if in.CookUserID != nil {
		out.CookUserID = *in.CookUserID
	}
	return out
}

func (s *Server) handleGetWeek(w http.ResponseWriter, r *http.Request) {
	week := r.PathValue("week")
	entries, err := s.deps.Plan.Week(r.Context(), principalFrom(r).HouseholdID, week)
	if err != nil {
		s.fail(w, r, err)
		return
	}
	out := make([]entryJSON, 0, len(entries))
	for _, e := range entries {
		out = append(out, toEntryJSON(e))
	}
	writeJSON(w, http.StatusOK, map[string]any{"week": week, "entries": out})
}

func (s *Server) handleAddEntry(w http.ResponseWriter, r *http.Request) {
	var in entryInJSON
	if !decode(w, r, &in) {
		return
	}
	e, err := s.deps.Plan.Add(r.Context(), principalFrom(r).HouseholdID, r.PathValue("week"), in.toInput())
	if err != nil {
		s.fail(w, r, err)
		return
	}
	writeJSON(w, http.StatusCreated, toEntryJSON(e))
}

func (s *Server) handleEditEntry(w http.ResponseWriter, r *http.Request) {
	var in entryInJSON
	if !decode(w, r, &in) {
		return
	}
	e, err := s.deps.Plan.Edit(r.Context(), principalFrom(r).HouseholdID, r.PathValue("week"), r.PathValue("id"), in.toInput())
	if err != nil {
		s.fail(w, r, err)
		return
	}
	writeJSON(w, http.StatusOK, toEntryJSON(e))
}

func (s *Server) handleRemoveEntry(w http.ResponseWriter, r *http.Request) {
	if err := s.deps.Plan.Remove(r.Context(), principalFrom(r).HouseholdID, r.PathValue("week"), r.PathValue("id")); err != nil {
		s.fail(w, r, err)
		return
	}
	w.WriteHeader(http.StatusNoContent)
}

func (s *Server) handleCopyWeek(w http.ResponseWriter, r *http.Request) {
	var in struct {
		FromWeek string `json:"fromWeek"`
	}
	if !decode(w, r, &in) {
		return
	}
	n, err := s.deps.Plan.Copy(r.Context(), principalFrom(r).HouseholdID, in.FromWeek, r.PathValue("week"))
	if err != nil {
		s.fail(w, r, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]int{"copied": n})
}
```

```go
package store

import (
	"context"
	"database/sql"

	"mealplan/internal/domain"
)

// AddMember creates a user and their membership atomically.
func (s *Store) AddMember(ctx context.Context, u domain.User, householdID string, role domain.Role) error {
	return s.inTx(ctx, func(tx *sql.Tx) error {
		if err := s.insertUser(ctx, tx, u); err != nil {
			return err
		}
		return s.exec(ctx, tx, `INSERT INTO membership (user_id, household_id, role) VALUES (?, ?, ?)`, u.ID, householdID, string(role))
	})
}

// adminsExcluding counts the household's admins other than userID.
func (s *Store) adminsExcluding(ctx context.Context, tx *sql.Tx, householdID, userID string) (int, error) {
	var n int
	err := tx.QueryRowContext(ctx, s.db.Rebind(
		`SELECT COUNT(*) FROM membership WHERE household_id = ? AND role = 'admin' AND user_id <> ?`), householdID, userID).Scan(&n)
	return n, err
}

func (s *Store) memberRole(ctx context.Context, tx *sql.Tx, householdID, userID string) (domain.Role, error) {
	var role string
	err := tx.QueryRowContext(ctx, s.db.Rebind(`SELECT role FROM membership WHERE household_id = ? AND user_id = ?`), householdID, userID).Scan(&role)
	if err != nil {
		return "", mapErr(err)
	}
	return domain.Role(role), nil
}

// SetRole changes a member's role; the household must keep at least one admin.
func (s *Store) SetRole(ctx context.Context, householdID, userID string, role domain.Role) error {
	return s.inTx(ctx, func(tx *sql.Tx) error {
		cur, err := s.memberRole(ctx, tx, householdID, userID)
		if err != nil {
			return err
		}
		if cur == domain.RoleAdmin && role != domain.RoleAdmin {
			others, err := s.adminsExcluding(ctx, tx, householdID, userID)
			if err != nil {
				return err
			}
			if others == 0 {
				return ErrLastAdmin
			}
		}
		return s.exec(ctx, tx, `UPDATE membership SET role = ? WHERE household_id = ? AND user_id = ?`, string(role), householdID, userID)
	})
}

// RemoveMember ends a membership: it deletes the user's sessions in the household and clears
// them as cook from the household's plan. The user account itself is kept.
func (s *Store) RemoveMember(ctx context.Context, householdID, userID string) error {
	return s.inTx(ctx, func(tx *sql.Tx) error {
		cur, err := s.memberRole(ctx, tx, householdID, userID)
		if err != nil {
			return err
		}
		if cur == domain.RoleAdmin {
			others, err := s.adminsExcluding(ctx, tx, householdID, userID)
			if err != nil {
				return err
			}
			if others == 0 {
				return ErrLastAdmin
			}
		}
		for _, q := range []string{
			`UPDATE plan_entry SET cook_user_id = NULL WHERE household_id = ? AND cook_user_id = ?`,
			`DELETE FROM session WHERE household_id = ? AND user_id = ?`,
			`DELETE FROM membership WHERE household_id = ? AND user_id = ?`,
		} {
			if err := s.exec(ctx, tx, q, householdID, userID); err != nil {
				return err
			}
		}
		return nil
	})
```

```go
// Package shopping turns a meal plan into a shopping list.
package shopping

import (
	"sort"

	"mealplan/internal/domain"
	"mealplan/internal/units"
)

// Line is one row of a generated list. A nil Quantity means "as needed" (no amount known).
type Line struct {
	Name           string
	NormalizedName string
	Quantity       *float64
	Unit           string
}

type acc struct {
	name, norm, unit string // unit is the base unit for quantified rows
	amount           float64
	hasAmount        bool
}

func key(norm, unit string, hasAmount bool) string {
	if !hasAmount {
		return norm + "|-"
	}
	return norm + "|" + unit
}

// Build scales every recipe entry's ingredients by servings, sums matching rows, subtracts stock and
// returns the rows sorted by name. Free-text entries and unknown recipes contribute nothing.
func Build(entries []domain.PlanEntry, recipes map[string]domain.Recipe, stock []Line) ([]Line, error) {
	rows := map[string]*acc{}
	for _, e := range entries {
		if e.RecipeID == "" {
			continue
		}
		r, ok := recipes[e.RecipeID]
		if !ok {
			continue
		}
		for _, ing := range r.Ingredients {
			if ing.Quantity == nil {
				k := key(ing.NormalizedName, "", false)
				if _, seen := rows[k]; !seen {
					rows[k] = &acc{name: ing.Name, norm: ing.NormalizedName}
				}
				continue
			}
			scaled, err := units.Scale(*ing.Quantity, e.Servings, r.BaseServings)
			if err != nil {
				return nil, err
			}
			amount, unit := units.ToBase(scaled, ing.Unit)
			k := key(ing.NormalizedName, unit, true)
			a, seen := rows[k]
			if !seen {
				a = &acc{name: ing.Name, norm: ing.NormalizedName, unit: unit, hasAmount: true}
				rows[k] = a
			}
			a.amount += amount
		}
	}

	for _, s := range stock {
		if s.Quantity == nil {
			continue
		}
		amount, unit := units.ToBase(*s.Quantity, s.Unit)
		if a, ok := rows[key(s.NormalizedName, unit, true)]; ok {
			a.amount -= amount
			if a.amount <= 0 {
				delete(rows, key(s.NormalizedName, unit, true))
			}
		}
	}

	out := make([]Line, 0, len(rows))
	for _, a := range rows {
		l := Line{Name: a.name, NormalizedName: a.norm}
		if a.hasAmount {
			v, u := units.Humanize(a.amount, a.unit)
			l.Quantity, l.Unit = &v, u
		}
		out = append(out, l)
	}
	sort.Slice(out, func(i, j int) bool {
		if out[i].NormalizedName != out[j].NormalizedName {
			return out[i].NormalizedName < out[j].NormalizedName
		}
		return out[i].Unit < out[j].Unit
	})
	return out, nil
}
```

```go
package store

import (
	"context"
	"database/sql"

	"mealplan/internal/domain"
)

const itemCols = `id, household_id, week, name, normalized_name, quantity, unit, checked, manual`

func b2i(b bool) int {
	if b {
		return 1
	}
	return 0
}

func qty(p *float64) any {
	if p == nil {
		return nil
	}
	return *p
}

func scanItem(r scanner) (domain.ShoppingItem, error) {
	var it domain.ShoppingItem
	var q sql.NullFloat64
	var checked, manual int
	if err := r.Scan(&it.ID, &it.HouseholdID, &it.Week, &it.Name, &it.NormalizedName, &q, &it.Unit, &checked, &manual); err != nil {
		return domain.ShoppingItem{}, mapErr(err)
	}
	if q.Valid {
		v := q.Float64
		it.Quantity = &v
	}
	it.Checked, it.Manual = checked != 0, manual != 0
	return it, nil
}

func (s *Store) insertItem(ctx context.Context, e execer, it domain.ShoppingItem) error {
	return s.exec(ctx, e, `INSERT INTO shopping_item (`+itemCols+`) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
		it.ID, it.HouseholdID, it.Week, it.Name, it.NormalizedName, qty(it.Quantity), it.Unit, b2i(it.Checked), b2i(it.Manual))
}

func (s *Store) InsertShoppingItem(ctx context.Context, it domain.ShoppingItem) error {
	return s.insertItem(ctx, s.db, it)
}

func (s *Store) ShoppingItems(ctx context.Context, householdID, week string) ([]domain.ShoppingItem, error) {
	rows, err := s.db.QueryContext(ctx, s.db.Rebind(
		`SELECT `+itemCols+` FROM shopping_item WHERE household_id = ? AND week = ? ORDER BY checked, normalized_name, unit, id`), householdID, week)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []domain.ShoppingItem
	for rows.Next() {
		it, err := scanItem(rows)
		if err != nil {
			return nil, err
		}
		out = append(out, it)
	}
	return out, rows.Err()
}

func (s *Store) ShoppingItemByID(ctx context.Context, householdID, id string) (domain.ShoppingItem, error) {
	return scanItem(s.db.QueryRowContext(ctx, s.db.Rebind(`SELECT `+itemCols+` FROM shopping_item WHERE id = ? AND household_id = ?`), id, householdID))
}

func (s *Store) UpdateShoppingItem(ctx context.Context, it domain.ShoppingItem) error {
	res, err := s.db.ExecContext(ctx, s.db.Rebind(
		`UPDATE shopping_item SET name = ?, normalized_name = ?, quantity = ?, unit = ?, checked = ? WHERE id = ? AND household_id = ?`),
		it.Name, it.NormalizedName, qty(it.Quantity), it.Unit, b2i(it.Checked), it.ID, it.HouseholdID)
	if err != nil {
		return mapErr(err)
	}
	if n, _ := res.RowsAffected(); n == 0 {
		return ErrNotFound
	}
	return nil
}

func (s *Store) DeleteShoppingItem(ctx context.Context, householdID, id string) error {
	res, err := s.db.ExecContext(ctx, s.db.Rebind(`DELETE FROM shopping_item WHERE id = ? AND household_id = ?`), id, householdID)
	if err != nil {
		return mapErr(err)
	}
	if n, _ := res.RowsAffected(); n == 0 {
		return ErrNotFound
	}
	return nil
}

// ReplaceGeneratedItems swaps all non-manual items of the week for the given ones, atomically.
func (s *Store) ReplaceGeneratedItems(ctx context.Context, householdID, week string, items []domain.ShoppingItem) error {
	return s.inTx(ctx, func(tx *sql.Tx) error {
		if err := s.exec(ctx, tx, `DELETE FROM shopping_item WHERE household_id = ? AND week = ? AND manual = 0`, householdID, week); err != nil {
			return err
		}
		for _, it := range items {
			it.HouseholdID, it.Week = householdID, week
			if err := s.insertItem(ctx, tx, it); err != nil {
				return err
			}
		}
		return nil
	})
}

// RecipesByIDs loads full recipes for the household; foreign and unknown ids are silently skipped.
func (s *Store) RecipesByIDs(ctx context.Context, householdID string, ids []string) (map[string]domain.Recipe, error) {
	out := map[string]domain.Recipe{}
	for _, id := range ids {
		r, err := s.RecipeByID(ctx, householdID, id)
		if err == ErrNotFound {
			continue
		}
		if err != nil {
			return nil, err
		}
		out[id] = r
	}
	return out, nil
}
```

```go
type PlanSource interface {
	PlanEntries(ctx context.Context, householdID, fromDate, toDate string) ([]domain.PlanEntry, error)
	RecipesByIDs(ctx context.Context, householdID string, ids []string) (map[string]domain.Recipe, error)
}

type ItemInput struct {
	Name     string
	Quantity *float64
	Unit     string
}

type Service struct {
	repo  Repo
	plans PlanSource
}

func New(repo Repo, plans PlanSource) *Service { return &Service{repo: repo, plans: plans} }

func invalid(format string, a ...any) error {
	return fmt.Errorf("%w: "+format, append([]any{domain.ErrInvalid}, a...)...)
}

func (in ItemInput) validate() (name, unit string, err error) {
	name = strings.TrimSpace(in.Name)
	unit = units.CanonUnit(in.Unit)
	if n := utf8.RuneCountInString(name); n == 0 || n > maxName {
		return "", "", invalid("name must be 1 to %d characters", maxName)
	}
	if utf8.RuneCountInString(unit) > maxUnitLen {
		return "", "", invalid("unit must be at most %d characters", maxUnitLen)
	}
	if in.Quantity != nil {
		v := *in.Quantity
		if math.IsNaN(v) || math.IsInf(v, 0) || v <= 0 || v > maxQuantity {
			return "", "", invalid("quantity must be greater than 0 and at most %v", maxQuantity)
		}
	}
	return name, unit, nil
}

func itemKey(norm, unit string, hasQty bool) string {
	if !hasQty {
		return norm + "|-"
	}
	return norm + "|" + unit
}

func sameQty(a, b *float64) bool {
	if a == nil || b == nil {
		return a == nil && b == nil
	}
	return units.Round(*a) == units.Round(*b)
}

func (s *Service) List(ctx context.Context, householdID, week string) ([]domain.ShoppingItem, error) {
	if _, err := plan.ValidWeek(week); err != nil {
		return nil, err
	}
	return s.repo.ShoppingItems(ctx, householdID, week)
}

// Generate rebuilds the generated part of the week's list from the plan. Manual items are kept, and
// a generated item whose key and quantity did not change keeps its id and checked state.
func (s *Service) Generate(ctx context.Context, householdID, week, from, to string) ([]domain.ShoppingItem, error) {
	dates, err := plan.WeekDates(week)
	if err != nil {
		return nil, err
	}
	lo, hi := dates[0], dates[6]
	if from != "" || to != "" {
		if from == "" || to == "" || from > to || from < lo || to > hi {
			return nil, invalid("from and to must be dates of the week, from not after to")
		}
		if _, err := plan.ValidWeek(week); err != nil {
			return nil, err
		}
		lo, hi = from, to
	}
	entries, err := s.plans.PlanEntries(ctx, householdID, lo, hi)
	if err != nil {
		return nil, err
	}
	ids := make([]string, 0, len(entries))
	seen := map[string]bool{}
	for _, e := range entries {
		if e.RecipeID != "" && !seen[e.RecipeID] {
			seen[e.RecipeID] = true
			ids = append(ids, e.RecipeID)
		}
	}
	recipes, err := s.plans.RecipesByIDs(ctx, householdID, ids)
	if err != nil {
		return nil, err
	}
	lines, err := Build(entries, recipes, nil) // pantry stock is wired in during Stage 3
	if err != nil {
		return nil, err
	}

	existing, err := s.repo.ShoppingItems(ctx, householdID, week)
	if err != nil {
		return nil, err
	}
	prev := map[string]domain.ShoppingItem{}
	for _, it := range existing {
		if !it.Manual {
			prev[itemKey(it.NormalizedName, it.Unit, it.Quantity != nil)] = it
		}
	}
	fresh := make([]domain.ShoppingItem, 0, len(lines))
	for _, l := range lines {
		it := domain.ShoppingItem{ID: auth.NewID(), Week: week, Name: l.Name, NormalizedName: l.NormalizedName, Quantity: l.Quantity, Unit: l.Unit}
		if old, ok := prev[itemKey(l.NormalizedName, l.Unit, l.Quantity != nil)]; ok {
			it.ID = old.ID
			it.Checked = old.Checked && sameQty(old.Quantity, l.Quantity)
		}
		fresh = append(fresh, it)
	}
	if err := s.repo.ReplaceGeneratedItems(ctx, householdID, week, fresh); err != nil {
		return nil, err
	}
	return s.repo.ShoppingItems(ctx, householdID, week)
}

func (s *Service) AddManual(ctx context.Context, householdID, week string, in ItemInput) (domain.ShoppingItem, error) {
	if _, err := plan.ValidWeek(week); err != nil {
		return domain.ShoppingItem{}, err
	}
	name, unit, err := in.validate()
	if err != nil {
		return domain.ShoppingItem{}, err
	}
	it := domain.ShoppingItem{ID: auth.NewID(), HouseholdID: householdID, Week: week, Name: name,
		NormalizedName: units.NormalizeName(name), Quantity: in.Quantity, Unit: unit, Manual: true}
	if err := s.repo.InsertShoppingItem(ctx, it); err != nil {
		return domain.ShoppingItem{}, err
	}
	return it, nil
}

func (s *Service) SetChecked(ctx context.Context, householdID, id string, checked bool) error {
	it, err := s.repo.ShoppingItemByID(ctx, householdID, id)
	if err != nil {
		return err
	}
	it.Checked = checked
	return s.repo.UpdateShoppingItem(ctx, it)
}

func (s *Service) Edit(ctx context.Context, householdID, id string, in ItemInput) (domain.ShoppingItem, error) {
	it, err := s.repo.ShoppingItemByID(ctx, householdID, id)
	if err != nil {
		return domain.ShoppingItem{}, err
	}
	if !it.Manual {
		return domain.ShoppingItem{}, invalid("generated items cannot be edited; change the plan or recipe instead")
	}
	name, unit, err := in.validate()
	if err != nil {
		return domain.ShoppingItem{}, err
	}
	it.Name, it.NormalizedName, it.Quantity, it.Unit = name, units.NormalizeName(name), in.Quantity, unit
	if err := s.repo.UpdateShoppingItem(ctx, it); err != nil {
		return domain.ShoppingItem{}, err
	}
	return it, nil
}

func (s *Service) Remove(ctx context.Context, householdID, id string) error {
	return s.repo.DeleteShoppingItem(ctx, householdID, id)
}
```

```ts
export function formatQuantity(q: number | null, unit: string): string {
  if (q === null) return ''
  const n = q.toLocaleString('pl-PL', { maximumFractionDigits: 3 })
```

```ts
import type { Slot } from './api/types'

const pad = (n: number) => String(n).padStart(2, '0')

function fmt(d: Date): string {
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
}

function parse(date: string): Date {
  const [y, m, d] = date.split('-').map(Number)
  return new Date(y, m - 1, d)
}

// mondayOf returns the Monday of the week containing d (weeks run Monday to Sunday).
export function mondayOf(d: Date): string {
  const day = (d.getDay() + 6) % 7
  return fmt(new Date(d.getFullYear(), d.getMonth(), d.getDate() - day))
}

export function shiftWeek(week: string, n: number): string {
  const d = parse(week)
  return fmt(new Date(d.getFullYear(), d.getMonth(), d.getDate() + 7 * n))
}

export function weekDates(week: string): string[] {
  const d = parse(week)
  return Array.from({ length: 7 }, (_, i) => fmt(new Date(d.getFullYear(), d.getMonth(), d.getDate() + i)))
}

export function formatDay(date: string): string {
  return parse(date).toLocaleDateString('pl-PL', { weekday: 'short', day: 'numeric', month: 'short' })
}
```

```ts
    },

      await this.load(this.week)
```

```ts
import { defineStore } from 'pinia'
import { api } from '../api/client'
import type { ShoppingItem } from '../api/types'

interface ListResponse {
  week: string
  items: ShoppingItem[]
}

export const useShopping = defineStore('shopping', {
  state: () => ({ week: '', items: [] as ShoppingItem[] }),
  getters: {
    open: (s) => s.items.filter((i) => !i.checked),
    done: (s) => s.items.filter((i) => i.checked),
  },
  actions: {
    async load(week: string) {
      const res = await api<ListResponse>('GET', `/shopping?week=${week}`)
      this.week = res.week
      this.items = res.items
    },
    async generate(week: string, range?: { from: string; to: string }) {
      const res = await api<ListResponse>('POST', '/shopping/generate', { week, ...range })
      this.week = res.week
      this.items = res.items
    },
    async addManual(name: string, quantity: number | null, unit: string) {
      const it = await api<ShoppingItem>('POST', '/shopping/items', { week: this.week, name, quantity, unit })
      this.items.push(it)
    },
    async setChecked(id: string, checked: boolean) {
      const it = this.items.find((i) => i.id === id)
      if (!it) return
      const before = it.checked
      it.checked = checked
      try {
        await api('PATCH', `/shopping/items/${encodeURIComponent(id)}`, { checked })
      } catch (e) {
        it.checked = before
        throw e
      }
    },
    async edit(id: string, input: { name: string; quantity: number | null; unit: string }) {
      await api('PATCH', `/shopping/items/${encodeURIComponent(id)}`, input)
      this.items = this.items.map((i) => (i.id === id ? { ...i, ...input } : i))
    },
    async remove(id: string) {
      await api('DELETE', `/shopping/items/${encodeURIComponent(id)}`)
      this.items = this.items.filter((i) => i.id !== id)
    },
  },
})
```
